```python
import cv2
import numpy as np

class SurveillancePipeline:
    def __init__(self, known_embeddings: dict[str, np.ndarray], threshold: float = 0.55):
        self.known_embeddings = known_embeddings
        self.threshold = threshold

    def extract_face_embeddings(self, face_frame: np.ndarray) -> np.ndarray:
        """Preprocesses frame and extracts normalized 128-d deep facial representation vector."""
        resized = cv2.resize(face_frame, (160, 160))
        normalized = resized.astype("float32") / 255.0
        # In actual deployment, passed into pre-trained FaceNet / ResNet embedding model
        embedding = np.random.randn(128)
        return embedding / np.linalg.norm(embedding)

    def match_suspect(self, candidate_vector: np.ndarray) -> tuple[str, float]:
        """Calculates Euclidean metric distance against verified criminal biometric records."""
        best_match, min_distance = "Unknown", float("inf")
        for criminal_id, record_vector in self.known_embeddings.items():
            distance = np.linalg.norm(candidate_vector - record_vector)
            if distance < min_distance and distance < self.threshold:
                best_match, min_distance = criminal_id, distance
        return best_match, min_distance
```
